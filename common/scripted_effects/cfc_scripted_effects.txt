cfc_prepare_safe_spawn = {
	# Hide vanilla's preferred FE-capital marker long enough to make the original
	# synth_queen_spawn effect select an empty, non-FE system instead.
	every_galaxy_planet = {
		limit = { has_carrier_flag = xenophobe_fe_capital }
		set_carrier_flag = cfc_restore_xenophobe_fe_capital
		remove_carrier_flag = xenophobe_fe_capital
	}

	# First preference: an entirely quiet system. The second pass only relaxes
	# the megastructure check; neither pass ever accepts an FE/AE system or a
	# system containing a colony.
	random_galaxy_planet = {
		limit = {
			pop_amount < 1
			NOR = {
				is_star = yes
				is_asteroid = yes
				is_planet_class = pc_broken
				is_planet_class = pc_shielded
			}
			solar_system = {
				NOT = { any_system_planet = { is_colony = yes } }
				NOT = { any_fleet_in_system = { is_mobile = yes } }
				has_any_megastructure = no
				OR = {
					NOT = { exists = owner }
					owner = { is_country_type = default }
				}
			}
		}
		save_global_event_target_as = cfc_spawn_bastille_candidate
	}
	if = {
		limit = { NOT = { exists = event_target:cfc_spawn_bastille_candidate } }
		random_galaxy_planet = {
			limit = {
				pop_amount < 1
				NOR = {
					is_star = yes
					is_asteroid = yes
					is_planet_class = pc_broken
					is_planet_class = pc_shielded
				}
				solar_system = {
					NOT = { any_system_planet = { is_colony = yes } }
					NOT = { any_fleet_in_system = { is_mobile = yes } }
					OR = {
						NOT = { exists = owner }
						owner = { is_country_type = default }
					}
				}
			}
			save_global_event_target_as = cfc_spawn_bastille_candidate
		}
	}
	if = {
		limit = { NOT = { exists = event_target:cfc_spawn_bastille_candidate } }
		# Pathological fully occupied galaxies get a tiny isolated fallback rather
		# than ever reverting to vanilla's forced FE-capital sacrifice.
		random_rim_system = {
			spawn_system = {
				min_distance >= 20
				max_distance <= 50
				initializer = "cfc_safe_bastille_system"
				effect = {
					random_system_planet = {
						limit = { has_carrier_flag = cfc_safe_bastille_world }
						save_global_event_target_as = cfc_spawn_bastille_candidate
					}
				}
			}
		}
	}

	if = {
		limit = { exists = event_target:cfc_spawn_bastille_candidate }
		event_target:cfc_spawn_bastille_candidate = {
			if = {
				limit = { NOT = { exists = owner } }
				set_owner = root
			}
			set_carrier_flag = xenophobe_fe_capital
		}
	}
}

cfc_restore_spawn_markers = {
	if = {
		limit = { exists = event_target:cfc_spawn_bastille_candidate }
		event_target:cfc_spawn_bastille_candidate = {
			remove_carrier_flag = xenophobe_fe_capital
			remove_carrier_flag = cfc_safe_bastille_world
		}
		clear_global_event_target = cfc_spawn_bastille_candidate
	}
	every_galaxy_planet = {
		limit = { has_carrier_flag = cfc_restore_xenophobe_fe_capital }
		set_carrier_flag = xenophobe_fe_capital
		remove_carrier_flag = cfc_restore_xenophobe_fe_capital
	}
}

cfc_normalize_fallen_empire = {
	# Both counters are read by cfc.30 every month, so they are initialized here,
	# before that event can ever run for this country.
	if = {
		limit = { NOT = { has_country_flag = cfc_fe_counters_ready } }
		set_country_flag = cfc_fe_counters_ready
		set_variable = { which = cfc_collapse_months value = 0 }
		set_variable = { which = cfc_peace_months value = 0 }
	}
	if = {
		limit = { has_modifier = beset_by_cetana }
		remove_modifier = beset_by_cetana
	}
	if = {
		limit = { has_country_flag = synth_queen_storm }
		deactivate_fog_machine = yes
		remove_country_flag = synth_queen_storm
	}
	remove_country_flag = weaker_navy
	if = {
		limit = { NOT = { has_country_flag = protected_from_queen_storm } }
		set_country_flag = protected_from_queen_storm
		log = "[Cetana Fair Crisis] Fleet scripted damage/destruction intercepted for [This.GetName]."
	}
	if = {
		limit = { NOT = { has_country_flag = cfc_fe_normalized } }
		set_country_flag = cfc_fe_normalized
		log = "[Cetana Fair Crisis] FE combat modifier normalized for [This.GetName]."
	}
}

# Shared by the player option in crisis.8063 and by the AI decision in cfc.40 so
# both routes grant exactly the same vanilla bypass flags and war.
cfc_join_early_cetana_war = {
	set_country_flag = cfc_joined_early_cetana_war
	set_country_flag = synth_queen_cannot_yeet_the_fleets
	set_country_flag = protected_from_queen_storm
	declare_war = {
		target = event_target:synth_queen_country_global
		name = cfc_cetana_intervention_war
		attacker_war_goal = cfc_wg_intervene_cetana
	}
	log = "[Cetana Fair Crisis] [This.GetName] declared war on Cetana during the FE phase."
}

# Cetana's conventional victory over one Fallen/Awakened Empire. It is only ever
# called after cfc_fallen_empire_is_militarily_broken has held for six
# consecutive months, so the outcome is produced by the war, not by a timer.
# The vanilla wipe effect is reused unchanged for the actual removal.
cfc_collapse_fallen_empire = {
	log = "[Cetana Fair Crisis] [This.GetName] lost the war against Cetana; vanilla system wipe applied."
	every_system_within_border = {
		synth_queen_wipe_system = yes
	}
}

# Diminishing replacement for the repeated vanilla crisis.8050 top-up. $CAP$ is
# the maximum number of mobile fleets this wave restores.
cfc_queen_reinforce = {
	while = {
		count = 15
		limit = {
			count_owned_fleet = {
				count < $CAP$
				limit = { is_mobile = yes }
			}
		}
		synth_queen_create_war_fleet = yes
	}
	log = "[Cetana Fair Crisis] Cetana reinforcement wave capped at $CAP$ fleets."
}

cfc_cleanup_early_phase = {
	every_country = {
		limit = {
			OR = {
				is_country_type = fallen_empire
				is_country_type = awakened_fallen_empire
			}
		}
		if = {
			limit = { has_modifier = beset_by_cetana }
			remove_modifier = beset_by_cetana
		}
		if = {
			limit = { has_country_flag = synth_queen_storm }
			deactivate_fog_machine = yes
		}
		remove_country_flag = synth_queen_storm
		remove_country_flag = weaker_navy
		remove_country_flag = protected_from_queen_storm
		remove_country_flag = cfc_fe_normalized
		if = {
			limit = { has_country_flag = cfc_fe_counters_ready }
			remove_country_flag = cfc_fe_counters_ready
			remove_country_flag = cfc_fe_tick_recent
			set_variable = { which = cfc_collapse_months value = 0 }
			set_variable = { which = cfc_peace_months value = 0 }
		}
		every_system_within_border = {
			if = {
				limit = { has_modifier = queen_small_scorn_storm }
				remove_modifier = queen_small_scorn_storm
			}
			if = {
				limit = { has_modifier = queen_scorn_storm }
				remove_modifier = queen_scorn_storm
			}
			if = {
				limit = { has_modifier = queen_big_scorn_storm }
				remove_modifier = queen_big_scorn_storm
			}
		}
	}
	every_country = {
		limit = { has_country_flag = cfc_joined_early_cetana_war }
		remove_country_flag = cfc_joined_early_cetana_war
		remove_country_flag = synth_queen_cannot_yeet_the_fleets
		remove_country_flag = protected_from_queen_storm
	}
	every_country = {
		limit = { has_country_flag = cfc_intervention_cooldown }
		remove_country_flag = cfc_intervention_cooldown
	}
	every_country = {
		limit = { has_country_flag = cfc_reinforcement_chain_active }
		remove_country_flag = cfc_reinforcement_chain_active
		set_variable = { which = cfc_reinforcement_wave value = 0 }
	}
}
